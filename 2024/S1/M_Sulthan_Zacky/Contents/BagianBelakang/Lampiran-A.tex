clc; clear all; close all;

% Equation statements
r = 0:0.1:20;
P = log(2)/2 .* ((r.^2 - 4) ./ (r.^3 - r.^2));

% Plots
plot(r, P, 'LineWidth', 2, 'Color', '0.85, 0.33, 0.10');
xlim([min(r), max(r)]);
ylim([0, 0.1]);
xlabel('r', 'FontWeight', 'bold');
ylabel('P*', 'FontWeight', 'bold');

clc; clear all; close all;

% Equation statements
r = 0:0.1:20;

B1 = 0;
eta1 = (1 - (2 ./ r).^2) ./ (1 + B1 .* (r - 1));

B2 = 0.01;
eta2 = (1 - (2 ./ r).^2) ./ (1 + B2 .* (r - 1));

B3 = 0.03;
eta3 = (1 - (2 ./ r).^2) ./ (1 + B3 .* (r - 1));

B4 = 0.07;
eta4 = (1 - (2 ./ r).^2) ./ (1 + B4 .* (r - 1));

% Plots
plot(r, eta1, 'DisplayName', '\beta = 0', 'LineWidth', 2, 'Color', ...
'0.85, 0.33, 0.10');
hold on
plot(r, eta2, 'DisplayName', '\beta = 0.01', 'LineWidth', 2, 'Color', ...
'0.93, 0.69, 0.13');
plot(r, eta3, 'DisplayName', '\beta = 0.03', 'LineWidth', 2, 'Color', ...
'0.39, 0.83, 0.07');
plot(r, eta4, 'DisplayName', '\beta = 0.07', 'LineWidth', 2, 'Color', ...
'0.30, 0.75, 0.93');

xlim([min(r), max(r)]);
ylim([0, max(eta1)]);
xlabel('r', 'FontWeight', 'bold');
ylabel('\eta', 'FontWeight', 'bold');
legend('show', 'Location', 'southeast');
